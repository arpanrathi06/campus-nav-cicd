pipeline {
    agent any
    environment {
        AWS_REGION = 'ap-south-1' // Your AWS region
        AWS_ACCOUNT_ID = '404571418135' // Replace with your 12-digit AWS Account ID
        REPO_NAME = 'campus-nav'
        IMAGE_TAG = 'latest'
        ECR_REGISTRY = "${AWS_ACCOUNT_ID}.dkr.ecr.${AWS_REGION}.amazonaws.com"
        IMAGE_URI = "${ECR_REGISTRY}/${REPO_NAME}:${IMAGE_TAG}"
    }
    stages {
        stage('1. Checkout Code') {
            steps { checkout scm }
        }
        stage('2. Build and Test') {
            steps { sh 'g++ -o campus_nav smart.cpp' }
        }
        stage('3. Docker Build') {
            steps {
                sh "docker build -t ${env.IMAGE_URI} ."
            }
        }
        stage('4. AWS ECR Login & Push') {
            steps {
                sh "aws ecr get-login-password --region ${env.AWS_REGION} | docker login --username AWS --password-stdin ${env.ECR_REGISTRY}"
                sh "docker push ${env.IMAGE_URI}"
            }
        }
       stage('5. Docker Deployment') {
            steps {
                // Stop and remove any old container if it exists
                sh 'docker stop campus-nav-app || true'
                sh 'docker rm campus-nav-app || true'
                
                // Run the fresh image from ECR
                sh 'docker run -d --name campus-nav-app -p 8080:8080 404571418135.dkr.ecr.ap-south-1.amazonaws.com/campus-nav:latest'
            }
        }
        stage('6. Verification') {
            steps {
                sh 'docker ps'
                sh 'docker logs campus-nav-app'
            }
        }
    }
}
