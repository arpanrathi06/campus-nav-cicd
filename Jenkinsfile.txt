pipeline {
    agent any
    environment {
        DOCKER_IMAGE = 'your-dockerhub-username/campus-nav:latest'
        REGISTRY_CREDENTIALS = 'dockerhub-credentials-id'
    }
    stages {
        stage('1. Checkout Code') {
            steps { checkout scm }
        }
        stage('2. Build and Test') {
            steps { sh 'g++ -o campus_nav main.cpp' }
        }
        stage('3. Docker Build') {
            steps {
                script { app = docker.build("${env.DOCKER_IMAGE}") }
            }
        }
        stage('4. Push Image') {
            steps {
                script {
                    docker.withRegistry('https://index.docker.io/v1/', "${env.REGISTRY_CREDENTIALS}") {
                        app.push("latest")
                    }
                }
            }
        }
        stage('5. Minikube Deployment') {
            steps {
                sh 'kubectl apply -f k8s/deployment.yaml'
                sh 'kubectl apply -f k8s/service.yaml'
            }
        }
        stage('6. Verification') {
            steps {
                sh 'kubectl get pods'
                sh 'kubectl get services'
            }
        }
    }
}
